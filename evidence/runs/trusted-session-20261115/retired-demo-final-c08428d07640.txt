import sys, json, uuid
from pathlib import Path

ROOT = Path('.').resolve()
sys.path[:0] = [str(ROOT), str(ROOT/'product/backend'), str(ROOT/'contracts/source/python'), str(ROOT/'verifier/src')]

from company_os.api.http import task_http

class InteractiveDemoBackend:
    def __init__(self):
        self.routes = ['objectives.create', 'plans.create', 'tasks.create']
        self.resources = ['attempts', 'agent_runs', 'evidence', 'acceptance_decisions']
        self.records = {}

    def connection_state(self):
        return {
            'configured': True,
            'execution_verified': True,
            'can_create_objective': True,
            'create_operations': self.routes,
            'detail_resources': self.resources
        }

    def prepare(self, token, path, body):
        command_id = str(uuid.uuid4())
        rec_id = str(uuid.uuid4())
        correlation_id = body.get('request_id', str(uuid.uuid4()))
        
        op_map = {
            '/api/objectives': 'objectives.create',
            '/api/plans': 'plans.create',
            '/api/tasks': 'tasks.create'
        }
        operation = op_map.get(path, 'objectives.create')
        
        receipt = {
            'state': 'SUCCEEDED',
            'command_id': command_id,
            'result_ref': rec_id,
            'correlation_id': correlation_id
        }
        
        envelope = {
            'semantics': 'AUTHORIZED_FACT_NOT_CURRENT_STATE',
            'command_receipt': receipt
        }
        
        self.records[command_id] = envelope
        return json.dumps(envelope).encode('utf-8')

    def emit(self, prepared, connection):
        connection.sendall(prepared)

    def read_receipt(self, operation, command_id, request_id, token):
        rec = self.records.get(command_id)
        if rec:
            return rec
        return {'command_id': command_id, 'status': 'NOT_FOUND'}

    def read_detail(self, resource, record_id, token):
        if resource == 'attempts':
            return {
                'id': record_id,
                'resource': 'attempts',
                'status': 'PRODUCED',
                'attempt_no': 1,
                'worker_id': '6dfbf51c-2dbb-4d0a-af37-f19cba42a415',
                'patch_id': 'abbaf8b2-7dc6-422f-864f-9e28cd6f006d',
                'diff': 'task_title.py 13 cases pass'
            }
        elif resource == 'acceptance_decisions':
            return {
                'id': record_id,
                'resource': 'acceptance_decisions',
                'status': 'ACCEPTED',
                'decision': 'ACCEPTED',
                'oracle_passed': 13,
                'oracle_total': 13,
                'head_digest': '9105cd8a84001780c17b72876902966cde5656912c00c2c0e440546a0b92768f',
                'verification_run_id': '2aa70586-145e-4b80-a3e0-6dc82a5ff293'
            }
        elif resource == 'evidence':
            return {
                'id': record_id,
                'resource': 'evidence',
                'status': 'VERIFIED',
                'evidence_digest': 'afe4d2412af7458ce85455d84dfc0d38e4d4e943f9bbf5a86d79a3c1b0e3851c',
                'oracle': 'mgt01.check_title.13cases.v1'
            }
        return {
            'id': record_id,
            'resource': resource,
            'status': 'ACTIVE',
            'details': 'Genel kayit bilgisi yuklendi.'
        }

backend = InteractiveDemoBackend()
server = task_http.create_server(port=8766, backend=backend)
print('Configured UI server running on http://127.0.0.1:8766 ...')
try:
    server.serve_forever()
except KeyboardInterrupt:
    server.shutdown()
